--- evaluation_projects Table SQL ---
CREATE TABLE "evaluation_projects" (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    evaluatee_id INTEGER NOT NULL,
                    title TEXT NOT NULL,
                    status TEXT DEFAULT 'active',
                    start_date TEXT,
                    end_date TEXT,
                    evaluation_type TEXT DEFAULT '동료사원 평가',
                    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                    FOREIGN KEY (evaluatee_id) REFERENCES employees (id) ON DELETE CASCADE,
                    UNIQUE(evaluatee_id, evaluation_type)
                )

--- evaluation_questions Table SQL ---
CREATE TABLE evaluation_questions (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            question_text TEXT NOT NULL,
            category TEXT NOT NULL,
            is_essay INTEGER DEFAULT 0,  -- 0: 객관식(5점 척도), 1: 주관식(서술형)
            is_active INTEGER DEFAULT 1,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        , sort_order INTEGER DEFAULT 0, question_sub_text TEXT DEFAULT '', evaluation_type TEXT DEFAULT '동료사원 평가')
